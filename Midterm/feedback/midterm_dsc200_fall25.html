The order of letters in the returned list should follow their first appearance in the input string.

def question1(s):
    '''
    input: s - string containing lowercase English letters
    output: list - unique letters that appear more than once,
                   ordered by first appearance
    Example: "programming" -> ['r', 'g', 'm']
    '''
    # YOUR CODE HERE
    dic = {}
    for i in range(len(s)):
        char = s[i]
        dic[char] = dic.get(char, 0) + 1

    res = []
    for k, v in dic.items():
        if v > 1:
            res += [k]

    return res

def question2(x):
    '''
    input: x - 2D NumPy array
    output: list of 1D arrays - for each row, elements greater than that row's mean
    Example: [[1,2,3],[4,5,6],[7,8,9]] -> [array([3]), array([6]), array([9])]
    '''
    # YOUR CODE HERE
    res = []
    for row in x:
        temp = []
        mean = np.mean(row)
        for v in row:
            if v > mean:
                temp += [v]
        res += [np.array(temp)]

    return res

EmployeeID (int): A unique identifier for each employee.
Name (str): Employee's full name.
Salary (float): Employee's salary.
PerformanceScore (float): Employee's performance score (out of 100).

Write a function func_employee that takes a command argument from {'average_salary', 'top_performer', 'high_performers'} and does the following based on the given command:

• 'average_salary': returns the average salary of all employees.
• 'top_performer': returns the name of the employee with the highest performance score.
• 'high_performers': returns a list of employee names whose performance score is above 90.

If the command is not among the pre-determined options, the function should return "Command not found".

def func_employee(data, command):
    """
    input:
        data - pandas DataFrame with columns ['EmployeeID', 'Name', 'Salary', 'PerformanceScore']
        command - one of ['average_salary', 'top_performer', 'high_performers']
    output:
        - 'average_salary': return mean salary
        - 'top_performer': return name of employee with highest PerformanceScore
        - 'high_performers': return list of names with PerformanceScore above 90
        -  otherwise return "Command not found"
    """
    # YOUR CODE HERE
    if command == "average_salary":
        return data["Salary"].mean()
    elif command == "top_performer":
        return data.sort_values(by="PerformanceScore", ascending=False)["Name"].iloc[0]
    elif command == "high_performers":
        return list(data[data["PerformanceScore"] > 90]["Name"])
    return "Command not found"

Bonus Question Write a function called handtype that takes a list of 5 integers representing the faces of 5 dies (each integer is between 1 and 6) and returns a string indicating the type of hand according to the following rules:

  • "Five of a Kind": if all five dies show the same number. (E.g., [2, 2, 2, 2, 2])
  • "Four of a Kind": if four dies show the same number. (E.g., [3, 6, 3, 3, 3])
  • "Straight": if the dies show a sequence of five consecutive numbers. (E.g., [5, 1, 3, 4, 2])
  • "Full House": if three dies show one number and the other two dies show another number. (E.g., [4, 4, 2, 4, 2])
  • "Three of a Kind": if three dies show the same number. (E.g., [6, 3, 6, 6, 1])
  • "Two Pair": if there are two different pairs of dies showing the same number. (E.g., [5, 4, 2, 2, 5])
  • "One Pair": if there is one pair of dies showing the same number. (E.g., [1, 1, 3, 4, 6])
  • "Bust": if none of the above conditions are met. (E.g., [1, 4, 3, 2, 6])
Example input to the function: dies = [2, 5, 5, 5, 2] Function's output (handtype(dies)): 'Full House'

def handtype(dies):
    '''
    input: dies - list of 5 integers between 1 and 6 (inclusive)
    output: string - one of:
        "Five of a Kind", "Four of a Kind", "Straight",
        "Full House", "Three of a Kind", "Two Pair",
        "One Pair", or "Bust"
    Example: [2, 5, 5, 5, 2] -> 'Full House'
    '''
    # YOUR CODE HERE
    dic = {}
    for card in dies:
        dic[card] = dic.get(card, 0) + 1

    hand = np.sort(list(dic.values()))
    if len(hand) == 1 and np.allclose(hand, [5], 0):
        return "Five of a Kind"
    elif len(hand) == 2:
        if np.allclose(hand, [1, 4], 0):
            return "Four of a Kind"
        elif np.allclose(hand, [2, 3], 0):
            return "Full House"
    elif len(hand) == 3:
        if np.allclose(hand, [1, 1, 3], 0):
            return "Thee of a Kind"
        elif np.allclose(hand, [1, 2, 2], 0):
            return "Two of a Kind"
    elif len(hand) == 4: 
        if np.allclose(hand, [1, 1, 1, 2], 0):
            return "One Pair"

    sorted_dies = np.sort(dies)
    v = sorted_dies[0]
    for v2 in sorted_dies[1:]:
        if v2 - v != 1:
            return "Bust"
        v = v2

    return "Straight"
